PHASE 3 — IMPORTANT: Fix Position Sizing (Account for Open Capital)
================================================================================

FILE: services/decision_engine.py — _execute_open() method

PROBLEM:
  capital_to_use = context["balance"] * (size_pct / 100.0)

  This uses total balance regardless of how much is already in open trades.
  With 2 positions open at 30% each, only 40% of capital is free — but the
  engine could still try to deploy another 30%, over-committing by 20%.

CHANGE: Calculate free capital before sizing

  # Sum up value deployed in open positions
  deployed = sum(
      p["entry"] * p["quantity"] for p in context.get("portfolio", [])
  )
  free_capital = context["balance"] - deployed
  capital_to_use = free_capital * (size_pct / 100.0)

  if capital_to_use < 10:   # minimum threshold
      decision["error"] = "Insufficient free capital"
      decision["action"] = "SKIP"
      self._log_decision(decision)
      return

  reclaim = False

  def run(self):
      if not self.is_market_hours():
          logger.info("[TradingLoop] Outside market hours. Skipping cycle.")
          return

      if self.is_paused():
          logger.info("[TradingLoop] Autonomous trading is PAUSED. Skipping cycle.")
          return

      # Validate data feed before making decisions
      from services.moomoo_service import moomoo_service
      if not moomoo_service.is_connected:
          logger.warning("[TradingLoop] Moomoo not connected. Attempting reconnect...")
          if not moomoo_service.connect():
              from services.notifications import notification_queue
              notification_queue.enqueue(
                  "⚠️ *Trade cycle skipped* — Moomoo connection failed.",
                  level="warning",
                  category="general"
              )
              return

      decision_engine.run_decision_cycle()
